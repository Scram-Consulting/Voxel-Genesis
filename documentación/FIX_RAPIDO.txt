// 1. AGREGAR AL INICIO DEL ARCHIVO (después de includes):
const int MAX_CHUNKS_PER_FRAME = 2;  // Ajustar: 1=GPU lenta, 4=GPU rápida

// 2. AGREGAR EN main() DESPUÉS DE glfwMakeContextCurrent(window):
glfwSwapInterval(1);  // V-Sync ON

// 3. MODIFICAR EL GAME LOOP:

// Al inicio del while:
int chunksThisFrame = 0;

// Donde procesas chunks, CAMBIAR:
// ANTES:
for (auto* chunk : chunksToProcess) {
    processChunk(chunk);
}

// DESPUÉS:
for (auto* chunk : chunksToProcess) {
    if (chunksThisFrame >= MAX_CHUNKS_PER_FRAME) break;
    processChunk(chunk);
    chunksThisFrame++;
}
