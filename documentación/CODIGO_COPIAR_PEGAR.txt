std::cout << "Pre-cargando texturas de items..." << std::endl;
for (int i = 1; i < 16; i++) {
    GLuint tex = g_textureManager->getItemTexture((BlockType)i);
    if (tex == 0) {
        std::cerr << "WARNING: Textura " << i << " no cargó" << std::endl;
    }
}

// 2. Agregar ANTES del game loop:

if (g_textureManager == nullptr) {
    std::cerr << "ERROR: TextureManager nullptr!" << std::endl;
    g_textureManager = new TextureManager();
    g_textureManager->initialize();
}

BiomeData biomeS = generator.getBiomeData(worldX, worldZ - 8, 64);
BiomeData biomeE = generator.getBiomeData(worldX + 8, worldZ, 64);
BiomeData biomeW = generator.getBiomeData(worldX - 8, worldZ, 64);

// Promediar parámetros (transición suave)
BiomeData biomeData = biomeCenter;
biomeData.continentalness = (biomeCenter.continentalness * 0.5f +
                             (biomeN.continentalness + biomeS.continentalness +
                              biomeE.continentalness + biomeW.continentalness) * 0.125f);

biomeData.temperature = (biomeCenter.temperature * 0.5f +
                        (biomeN.temperature + biomeS.temperature +
                         biomeE.temperature + biomeW.temperature) * 0.125f);

biomeData.humidity = (biomeCenter.humidity * 0.5f +
                     (biomeN.humidity + biomeS.humidity +
                      biomeE.humidity + biomeW.humidity) * 0.125f);

biomeData.peaks = (biomeCenter.peaks * 0.5f +
                  (biomeN.peaks + biomeS.peaks +
                   biomeE.peaks + biomeW.peaks) * 0.125f);

// Mantener el bioma dominante del centro
biomeData.biomeType = biomeCenter.biomeType;
